FROM golang:1.22-alpine AS builder
WORKDIR /app
COPY Server/ ./Server/
WORKDIR /app/Server
RUN go mod download && CGO_ENABLED=0 GOOS=linux go build -o /app/server .
FROM alpine:3.19
RUN apk --no-cache add ca-certificates curl
WORKDIR /app
COPY --from=builder /app/server .
ENV PORT=8080
EXPOSE 8080
HEAL THCHECK --interval=30s --timeout=5s --start-period=10s --retries=3 CMD curl -f http://localhost:8080/health || exit 1
CMD ["./server"]
